\chapter{\iti{Essere} and \iti{Avere}} \label{M2-L1}

\begin{table}[h]
    \centering
    \begin{tabular}{r l}
      Person & Form \\
      \hline
      io      & \iti{sono}  \\
      tu      & \iti{sei}   \\
      lui/lei & \iti{è}     \\
      noi     & \iti{siamo} \\
      voi     & \iti{siete} \\
      loro    & \iti{sono} \\
    \end{tabular}
\end{table}

\section{\iti{Avere}: to have}
\begin{table}[h]
    \centering
    \begin{tabular}{r l}
        Person & Form \\
        \hline
        io      & \iti{ho}  \\
        tu      & \iti{hai}   \\
        lui/lei & \iti{ha}     \\
        noi     & \iti{abbiamo} \\
        voi     & \iti{avete} \\
        loro    & \iti{hanno} \\
    \end{tabular}
\end{table}

The H in \iti{ho, hai, ha, hanno} is \textbf{always} silent.

\section{\iti{Avere} for feelings}

\textbf{Warning!!} You \textit{must} use \iti{avere} with feelings, \textit{not} \iti{essere}!

\begin{table}[h]
    \centering
    \begin{tabular}{r l}
        Italian & English \\
        \hline
        \iti{ho fame}   & I am hungry  \\
        \iti{ho sete}   & I am thirsty \\
        \iti{ho freddo} & I am cold    \\
        \iti{ho caldo}  & I am hot     \\
        \iti{ho paura}  & I am afraid  \\
        \iti{ho sonno}  & I am sleepy  \\
        \iti{ho ... anni} & I am ... years old \\
    \end{tabular}
\end{table}

\textbf{Warning!!} \iti{Sono caldo} means being sexually hot, \textit{very} different from the temperature hot of \iti{ho caldo}!


\chapter{Regular Verbs: Present Tense} \label{M2-L2}

There are four `anchor' endings, common to all verb families.

\begin{itemize}[nosep]
    \item \iti{io} → \itb{-o}
    \item \iti{tu} → \itb{-i}
    \item \iti{noi} → \itb{-iamo}
    \item \iti{loro} → \itb{-no}
\end{itemize}

\section{Model Verbs conjugations}
\begin{table}[h]
    \centering
    \begin{tabular}{r | c l}
        Type & Model verb & Present tense \\
        \hline
        \textbf{-are}         & \iti{parlare} & \iti{parlo, parli, parla, parliamo, parlate, parlano} \\
        \textbf{-ere}         & \iti{leggere} & \iti{leggo, leggi, legge, leggiamo, leggete, leggono} \\
        \textbf{-ire}         & \iti{dormire} & \iti{dormo, dormi, dorme, dormiamo, dormite, dormono} \\
        \textbf{-ire + -isc-} & \iti{capire}  & \iti{capisco, capisci, capisce, capiamo, capite, capiscono} \\
    \end{tabular}
\end{table}

Notice that \iti{noi} and \iti{voi} \textbf{never} take the \iti{-isc-} insertion.

\section{Common model verbs}
\begin{itemize}[nosep]
    \item \itb{-are}: \iti{mangiare, lavorare, abitare, ascoltare, comprare, camminare}
    \item \itb{-ere}: \iti{vedere, scrivere, prendere, vivere, credere, chiudere}
    \item \itb{-ire}: \iti{partire, sentire, aprire, seguire}
    \item \itb{-ire + -isc-}: \iti{finire, preferire, pulire}
\end{itemize}

\chapter{Present Progressive: \iti{stare} + gerund} \label{M2-L4}

Used for actions happening \textbf{right now}: \iti{sto mangiando}, I am eating.

\section{Forming the Gerund}
\begin{table}[h]
    \centering
    \begin{tabular}{ r | c l }
        Verb family & Gerund ending & Example \\
        \hline
        \itb{-are} & \iti{-ando} & \iti{parlare} → \iti{parlando} \\
        \itb{-ere} & \iti{-endo} & \iti{leggere} → \iti{leggendo} \\
        \itb{-ire} & \iti{-endo} & \iti{dormire} → \iti{dormendo} \\
    \end{tabular}
\end{table}


\section{Irregular Gerunds}
\begin{itemize}[nosep]
    \item \iti{fare} → \itb{facendo}
    \item \iti{bere} → \itb{bevendo}
    \item \iti{dire} → \itb{dicendo}
\end{itemize}

\section{Structure}
Simply conjugate \iti{stare} and add the gerund.

\begin{itemize}[nosep]
    \item \iti{Sto mangiando.}    \\
    I am eating.
    \item \iti{Sta dormendo.}     \\
    He/she is sleeping.
    \item \iti{Stanno scrivendo.} \\
    They are writing.
\end{itemize}

Italian uses the present progressive \textbf{less} than English. Simple present tense is
often more natural for habits and near-future plans. For example, \iti{andare} almost never appears in the progressive.
