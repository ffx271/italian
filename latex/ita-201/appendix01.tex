\chapter{Quick Reference: Common Pitfalls}
\begin{itemize}
  \item \textbf{Mistake} \sout{struck through!}   → \textbf{should be...} → \textbf{...explanation}
    \\
    \item \sout{sono caldo}   → \itb{ho caldo} → Feelings use \iti{avere}
    \item \sout{sono 20 anni} → \itb{ho 20 anni} → Age uses \iti{avere}
    \item \sout{non mai vado} → \itb{non vado mai} → \iti{Non} always before verb
    \item \sout{sono uscite} → \itb{si sono alzate} → Reflexive needs pronoun
    \item \sout{abbiamo andato} → \itb{siamo andati} → \iti{Andare} uses \iti{essere}
    \item \sout{è dodici} → \itb{sono le dodici} → Time uses \iti{sono le}
    \item \sout{è l'uno} → \itb{è l'una} → One o'clock is feminine
    \item \sout{il quindici di marzo} → \itb{il quindici marzo} → No \iti{di} in dates
    \item \sout{in la nostra scuola} → \itb{nella nostra scuola} → \iti{In + la = nella}
    \item \sout{ogni giorni} → \itb{ogni giorno} → \iti{ogni} + singular
    \item \sout{arancioni} → \itb{arancione} → Invariable colour
    \item \sout{simpatichi} → \itb{simpatici} → No H in masculine plural
    \item \sout{le notte} → \itb{le notti} → -E nouns → -I in plural
    \item \sout{gl'amici} → \itb{gli amici} → \iti{Gli} never contracts
    \item \sout{voglio lo vedere} → \itb{voglio vederlo} → Pronoun attaches to infinitive
    \item \sout{ha venuto} → \itb{è venuto} → \iti{Venire} uses \iti{essere}
    \item \sout{era bellissima} (for completed year) → \itb{è stata bellissima} → Perfect for completed periods
\end{itemize}
